def bimodal_fusion(sign_res, veh_res):
    detections = []

    if sign_res[0].boxes is not None:
        for b in sign_res[0].boxes:
            detections.append({
                "box": b.xyxy.cpu().numpy()[0],
                "conf": float(b.conf),
                "label": "Traffic Sign"
            })

    if veh_res[0].boxes is not None:
        for b in veh_res[0].boxes:
            detections.append({
                "box": b.xyxy.cpu().numpy()[0],
                "conf": float(b.conf),
                "label": "Vehicle"
            })

    return detections
 

fused = bimodal_fusion(sign_res, veh_res)
print("Total fused detections:", len(fused))
